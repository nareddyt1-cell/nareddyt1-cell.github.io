export default {
  async fetch(request) {
    // ---- CORS preflight ----
    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders() });
    }

    // ---- Health check (Preview tab uses GET) ----
    if (request.method === "GET") {
      return new Response(
        JSON.stringify({
          status: "OK",
          message: "Enzyme Damage Analysis API is running. Use POST to analyze sequences."
        }, null, 2),
        { headers: { "Content-Type": "application/json", ...corsHeaders() } }
      );
    }

    // ---- Only POST below ----
    if (request.method !== "POST") {
      return new Response("Method Not Allowed", { status: 405 });
    }

    try {
      const { reference_sequence, damaged_sequence } = await request.json();

      if (!reference_sequence || !damaged_sequence) {
        return jsonResponse(
          { error: "reference_sequence and damaged_sequence are required" },
          400
        );
      }

      const wt = reference_sequence.replace(/\s+/g, "");
      const damaged = damaged_sequence.replace(/\s+/g, "");

      // ---- UniProt search ----
      const uniprotURL =
        "https://rest.uniprot.org/uniprotkb/search?query=" +
        encodeURIComponent(wt) +
        "&format=json&size=1";

      const uniRes = await fetch(uniprotURL);
      const uniData = await uniRes.json();

      if (!uniData.results || uniData.results.length === 0) {
        return jsonResponse({
          domains: [],
          analysis: {
            consequence: "No UniProt match found for the reference sequence."
          }
        });
      }

      const entry = uniData.results[0];
      const accession = entry.primaryAccession;

      // ---- Domains ----
      const domains = (entry.features || [])
        .filter(f => ["DOMAIN", "SIGNAL", "PROPEPTIDE"].includes(f.type))
        .map(f => ({
          name: f.description || f.type,
          start: f.location.start.value,
          end: f.location.end.value
        }));

      // ---- Catalytic damage logic ----
      let catalyticDamage = 0;
      for (let i = 50; i <= 250; i++) {
        if (wt[i] !== damaged[i]) catalyticDamage++;
      }

      const consequence =
        catalyticDamage === 0
          ? "No functional damage detected."
          : "Damage detected in catalytic region; reduced enzymatic activity predicted.";

      return jsonResponse({
        uniprot_accession: accession,
        alphafold_id: `AF-${accession}-F1`,
        alphafold_url: `https://alphafold.ebi.ac.uk/entry/AF-${accession}-F1`,
        domains,
        analysis: { consequence }
      });

    } catch (err) {
      return jsonResponse(
        { error: "Internal server error", details: err.message },
        500
      );
    }
  }
};

// ---------- helpers ----------
function corsHeaders() {
  return {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "POST, GET, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type"
  };
}

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data, null, 2), {
    status,
    headers: { "Content-Type": "application/json", ...corsHeaders() }
  });
}
