<script>
const API_URL = "https://enzyme-damage-api.YOURNAME.workers.dev";
const MOLSTAR_URL = "https://alphafold.ebi.ac.uk/files";

function clean(seq) {
  return seq.replace(/\s+/g, "").toUpperCase();
}

function highlight(ref, dmg) {
  let out = "";
  let diffCount = 0;

  const max = Math.max(ref.length, dmg.length);
  for (let i = 0; i < max; i++) {
    const r = ref[i] || "-";
    const d = dmg[i] || "-";
    if (r !== d) {
      diffCount++;
      out += `<span class="diff">${d}</span>`;
    } else {
      out += d;
    }
  }
  return { out, diffCount };
}

async function analyze() {
  const ref = clean(document.getElementById("ref").value);
  const dmg = clean(document.getElementById("dmg").value);

  if (!ref || !dmg) return alert("Both sequences are required.");

  const res = await fetch(API_URL, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      reference_sequence: ref,
      damaged_sequence: dmg
    })
  });

  const data = await res.json();

  // Domains
  const domBox = document.getElementById("domains");
  domBox.innerHTML = "";

  (data.domains || []).forEach(d => {
    domBox.innerHTML += `
      <div class="box amber">
        <strong>${d.name}</strong><br/>
        Residues ${d.start}–${d.end}<br/>
        <small>${d.description}</small>
      </div>
    `;
  });

  // Differences
  const diff = highlight(ref, dmg);
  document.getElementById("diffSeq").innerHTML = diff.out;
  document.getElementById("diffBox").classList.remove("hidden");

  // Biological consequence
  document.getElementById("bioText").innerText =
    diff.diffCount === 0
      ? "No functional differences detected."
      : "Detected differences may alter folding, stability, or enzymatic activity depending on affected regions.";

  document.getElementById("bioBox").classList.remove("hidden");

  // Comparative insight
  document.getElementById("compareText").innerText =
    diff.diffCount === 0
      ? "No damage-related differences detected."
      : `${diff.diffCount} damage-related differences detected.`;

  document.getElementById("compareBox").classList.remove("hidden");

  // Structure
  if (data.alphafold_id) {
    document.getElementById("viewer").src =
      `${MOLSTAR_URL}/${data.alphafold_id}-model_v4.pdb`;
    document.getElementById("structureBox").classList.remove("hidden");
  }
}
</script>
